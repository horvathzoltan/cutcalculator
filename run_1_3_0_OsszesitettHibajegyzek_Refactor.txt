Nagyon jó, hogy most erre fókuszálunk — ez az a pont, ahol a rendszered
„egydarabos termék‑számítóból” átlép egy valódi gyártási rendelés
feldolgozó pipeline‑ba.
A kódot átnéztem a teljes kontextusban, és most már pontosan látom,
hogyan kell a meglévő architektúrába beilleszteni az Order → OrderNeed →
WorksheetNeed folyamatot.

Az alábbi terv frissített, kód‑tudatos, integráció‑biztos, és teljesen
illeszkedik a jelenlegi projektstruktúrához.

------------------------------------------------------------------------

🚀 OrderNeed + WorksheetNeed – végleges, kódra szabott implementációs terv

🎯 Cél (változatlan, de pontosítva a kód alapján)

A rendszer tudja:

1.  Egy teljes order (több order item, qty‑vel) anyagszükségletét
    kiszámolni.
2.  Példány szinten: minden darab külön ItemNeed.
3.  Worksheet szinten aggregálni:
    -   Cutting list (összevont vágási tételek)
    -   Kitting list (összevont darabos tételek)

A pipeline:

    Order → OrderNeed → WorksheetNeed

A NeedCalculator már kész, csak fel kell építeni köré a két új réteget.

2.2 order_need_builder.cpp

Implementáció:

    #include "order_need_builder.h"

    OrderNeed OrderNeedBuilder::build(const Order& order, bool debug)
    {
        OrderNeed out;
        out.orderId = order.orderId;

        for (const auto& item : order.items) {
            OrderItemNeed oin;
            oin.orderItemId = item.orderItemId;
            oin.line = item.line;
            oin.itemNeeds.reserve(item.qty);

            for (int i = 0; i < item.qty; ++i) {
                ItemNeed n = NeedCalculator::calculate(item.line, item.modeName, debug);
                oin.itemNeeds.append(n);
            }

            out.orderItemNeeds.append(oin);
        }

        return out;
    }

        QUuid worksheetId;
        QVector<CutAggregatedItem> cuts;
        QVector<KitAggregatedItem> kits;
    };

    class WorksheetBuilder {

    };

4.2 worksheet_builder.cpp

    #include "worksheet_builder.h"
    #include "piece.h"
    #include "kit_item.h"

    WorksheetNeed WorksheetBuilder::fromOrderNeed(const OrderNeed& orderNeed)
    {
        WorksheetNeed ws;
        ws.worksheetId = orderNeed.orderId;

        QVector<Piece> pieces;
        QVector<KitItem> kits;

        for (const auto& oin : orderNeed.orderItemNeeds) {
            const OrderLine& line = oin.line;

            for (const auto& itemNeed : oin.itemNeeds) {

                // --- Cutting ---
                for (const RawCut& rc : itemNeed.cutItems) {
                    Piece p;
                    p.materialBarcode = rc.materialBarcode;
                    p.requiredLength  = rc.requiredLength;
                    p.handlerSide     = line.handlerSide;
                    p.externalRef     = line.externalId;
                    p.ownerName       = line.ownerName;
                    p.colorName       = line.colorName;
                    p.fullWidth       = line.width_mm;
                    p.fullHeight      = line.height_mm;
                    pieces.append(p);
                }

                // --- Kitting ---
                for (const RawKit& rk : itemNeed.kitItems) {
                    KitItem ki;
                    ki.materialBarcode = rk.materialBarcode;
                    ki.quantity        = rk.qty;
                    ki.ownerName       = line.ownerName;
                    ki.colorName       = line.colorName;
                    ki.fullWidth       = line.width_mm;
                    ki.fullHeight      = line.height_mm;
                    kits.append(ki);
                }
            }
        }

        ws.cuts = CutAggregator::aggregate(pieces);
        ws.kits = KitAggregator::aggregate(kits);

        return ws;
    }

5. OrderNeedBuilder teszt megírása

Fájl: test/calculation/order_need_builder_tester.cpp

5.1 Teszt skeleton

    #include "order_need_builder_tester.h"
    #include "calculation/service/order_need_builder.h"
    #include "calculation/service/worksheet_builder.h"
    #include "test/calculation/test_roletta_builder.h"

    bool OrderNeedBuilderTester::run()
    {
        zInfo("=== OrderNeedBuilder TEST START ===");

        // 1) Roletta testdata
        auto ids = TestRolettaBuilder::build();

        // 2) Order összeállítása
        Order order;
        order.orderId = QUuid::createUuid();

        OrderItem i1;
        i1.orderItemId = QUuid::createUuid();
        i1.line.productId = ids.ROL_R;
        i1.line.width_mm = 1200;
        i1.line.height_mm = 1500;
        i1.line.handlerSide = "L";
        i1.line.externalId = "X";
        i1.line.ownerName = "Teszt Elek";
        i1.line.colorName = "Fehér";
        i1.qty = 2;
        i1.modeName = "Manufacturing";

        order.items.append(i1);

        // 3) Build OrderNeed
        OrderNeed need = OrderNeedBuilder::build(order, true);

        Q_ASSERT(need.orderItemNeeds.size() == 1);
        Q_ASSERT(need.orderItemNeeds[0].itemNeeds.size() == 2);

        // 4) WorksheetNeed
        WorksheetNeed ws = WorksheetBuilder::fromOrderNeed(need);

        Q_ASSERT(!ws.cuts.isEmpty());
        Q_ASSERT(!ws.kits.isEmpty());

        zInfo("=== OrderNeedBuilder TEST END ===");
        return true;
    }
