🔧 1. LÉPÉS — ensureEmptyRow() (ha nincs, létrehozza)

FÁJL

order_item_table.cpp

ÚJ / VÉGLEGES VERZIÓ:

    void OrderItemTable::ensureEmptyRow()
    {
        int last = rowCount() - 1;

        // Ha nincs sor → hozzunk létre egyet
        if (last < 0) {
            addEmptyRow();
            return;
        }

        // Ha az utolsó sor nem üres → hozzunk létre egyet
        bool isEmpty = true;
        for (int col = 0; col < columnCount(); ++col) {
            auto* w = cellWidget(last, col);
            if (w && !w->property("empty").toBool()) {
                isEmpty = false;
                break;
            }
        }

        if (!isEmpty)
            addEmptyRow();
    }

🔧 2. LÉPÉS — addEmptyRow() (új üres sor létrehozása)

VÉGLEGES VERZIÓ:

    void OrderItemTable::addEmptyRow()
    {
        int row = rowCount();
        insertRow(row);

        // minden cellába üres widget + empty flag
        for (int col = 0; col < columnCount(); ++col) {
            QWidget* w = createEmptyCellWidget(col);
            w->setProperty("empty", true);
            setCellWidget(row, col, w);
        }
    }

A createEmptyCellWidget(col) a te meglévő cell‑widget gyárad.

------------------------------------------------------------------------

🔧 3. LÉPÉS — keyPressEvent() (Enter → új sor, Delete → törlés + auto‑empty row)

VÉGLEGES VERZIÓ:

    void OrderItemTable::keyPressEvent(QKeyEvent* e)
    {
        if (e->key() == Qt::Key_Return || e->key() == Qt::Key_Enter) {
            addEmptyRow();
            return;
        }

        if (e->key() == Qt::Key_Delete) {
            auto items = selectedItems();
            if (!items.isEmpty()) {
                int row = items.first()->row();
                deleteRowAt(row);
                ensureEmptyRow();
                return;
            }
        }

        QTableWidget::keyPressEvent(e);
    }

    void OrderItemTable::setItems(const QVector<OrderItem>& items)

        setRowCount(0);
